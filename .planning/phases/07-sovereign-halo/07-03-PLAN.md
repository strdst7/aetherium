
     - Initialize `attempts = 1`
     - Loop:
       ```typescript
       const report = await this.sovereignHalo.validate(responseText, identity, attempts);
       if (report.status === 'passed') {
         return { responseText, validationReport: report, attempts };
       }
       if (attempts >= MAX_HALO_ATTEMPTS) {
         const failureReport = this.sovereignHalo.generateFailureReport(report, attempts, identity);
         responseText = failureReport.safeFallbackMessage;
         return { responseText, validationReport: report, attempts };
       }
       // Tighten constraints and regenerate
       const violations = report.checks.filter(c => !c.passed).map(c => c.detail);
       const tightenedPrompt = this.buildTightenedPrompt(systemPrompt, memories, queryText, violations, attempts);
       const newTemp = Math.max(0.1, (req.temperature || 0.7) - (attempts * 0.1));
       const refined = await provider.generate({
         model: 'demo',
         prompt: tightenedPrompt,
         maxTokens: req.maxTokens || 200,
         temperature: newTemp,
       });
       responseText = refined.text || '';
       attempts++;
       ```
   - Step D: If `sovereignHalo` is undefined OR `identity` is null, return `{ responseText, attempts: 0 }` (preserve Phase 6 behavior)

4. Implement `buildTightenedPrompt()`:
   ```typescript
   private buildTightenedPrompt(
     systemPrompt: string,
     memories: any[],
     queryText: string,
     violations: string[],
     attemptNumber: number
   ): string {
     const constraintBlock = violations.map(v => `- ${v}`).join("\n");
     const doNotBlock = violations
       .filter(v => v.toLowerCase().includes("must not contain") || v.toLowerCase().includes("forbidden"))
       .map(v => `DO NOT: ${v}`)
       .join("\n");
     return `${systemPrompt}\n\n[Attempt ${attemptNumber}]\nPrevious attempt failed these checks:\n${constraintBlock}\n${doNotBlock ? "\nStrictly avoid:\n" + doNotBlock : ""}\n\nMemories:\n${JSON.stringify(memories)}\nUser:\n${queryText}`;
   }
   ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Provider.generate() → Orchestrator | LLM output is untrusted until validated |
| Orchestrator → Caller | Only validated (or safely failed) output crosses |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-07-08 | Spoofing | safeFallbackMessage injection | mitigate | Static message from identity config or hardcoded default; never raw LLM output |
| T-07-09 | Tampering | regeneration loop counter | mitigate | MAX_HALO_ATTEMPTS is a constant; attempts variable is local and incremented only after a new generation is received |
| T-07-10 | Denial of Service | infinite regeneration | mitigate | Bounded at 3 attempts; loop terminates unconditionally after max |
| T-07-11 | Information Disclosure | tightenedPrompt leaks violations | accept | Violations are user-defined identity rules, not secrets; they are meant to be enforced |
| T-07-12 | Elevation of Privilege | skip validation when identity is null | accept | Null identity means anonymous/unauthenticated request; skipping validation is correct behavior |
</threat_model>
