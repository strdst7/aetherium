     type: object

       passed: { type: boolean, description: "Whether the check passed" }
       detail: { type: string, description: "Human-readable result description" }
       confidence: { type: number, format: float, minimum: 0, maximum: 1, description: "Confidence score" }
       metadata: { type: object, description: "Optional extra data" }

   ValidationReport:
     type: object
     required: [status, checks, passedCount, failedCount, totalCount, confidenceScore, identityId, validatedAt]
     properties:
       status: { type: string, enum: [passed, failed], description: "Overall validation status" }
       checks: { type: array, items: { $ref: '#/components/schemas/ValidationCheck' } }
       passedCount: { type: integer, description: "Number of passed checks" }
       failedCount: { type: integer, description: "Number of failed checks" }
       totalCount: { type: integer, description: "Total number of checks" }
       confidenceScore: { type: number, format: float, minimum: 0, maximum: 1, description: "Aggregate confidence" }
       identityId: { type: string, description: "Identity this validation was for" }
       validatedAt: { type: string, format: date-time, description: "Validation timestamp" }
       attemptNumber: { type: integer, description: "Regeneration attempt number" }

   FailureReport:
     type: object
     required: [status, attemptCount, violationSummary, safeFallbackMessage, lastValidationReport, identityId, generatedAt]
     properties:
       status: { type: string, enum: [failed], description: "Failure status" }
       attemptCount: { type: integer, description: "Number of regeneration attempts" }
       violationSummary: { type: array, items: { type: string }, description: "Persistent violations" }
       safeFallbackMessage: { type: string, description: "Safe message for caller" }
       lastValidationReport: { $ref: '#/components/schemas/ValidationReport' }
       identityId: { type: string, description: "Identity this failure was for" }
       generatedAt: { type: string, format: date-time, description: "Failure timestamp" }
