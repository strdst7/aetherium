3. Add `processWithTools` method to `Orchestrator`:
```typescript
async processWithTools(req: OrchestratorRequestWithTools): Promise<OrchestratorResponse> {
  const { tools, toolExecutor, maxToolIterations = 5 } = req;
  
  // Step 1: Memory retrieval and prompt building (same as process())
  const { identity_anchor, messages, policy = {}, memoryAlpha = 0.5, memoryK = 5 } = req;
  const queryText = messages.map((m: any) => m.content).join(' ');
  
  const provider = await this.providerRegistry.pick({ ...policy, requireToolUse: true });
  
  let embedding: number[] = [0];
  if (provider.embed) {
    const embedResult = await provider.embed(queryText);
    embedding = Array.isArray(embedResult.embeddings[0]) 
      ? (embedResult.embeddings[0] as number[]) 
      : (embedResult.embeddings as number[]);
  }
  
  const memories = await this.memoryService.vectorSearch(embedding, memoryAlpha, memoryK);
  
  const systemPrompt = 'System: You are Aetherium, preserve identity fidelity.';
  const prompt = `${systemPrompt}\nMemories:\n${JSON.stringify(memories)}\nUser:\n${queryText}`;
  
  // Step 2: Initial generation with tools
  if (!provider.generateWithTools) {
    throw new Error('Provider does not support tool use');
  }
  
  let candidate = await provider.generateWithTools({
    model: 'gemini-1.5-pro',
    prompt,
    maxTokens: req.maxTokens,
    temperature: req.temperature
  }, tools);
  
  const toolExecutionTrace: ToolExecutionTrace[] = [];
  let iteration = 0;
  
  // Step 3: Tool loop
  while (candidate.toolCalls && candidate.toolCalls.length > 0 && iteration < maxToolIterations) {
    iteration++;
    
    for (const toolCall of candidate.toolCalls) {
      const result = await toolExecutor(toolCall);
      toolExecutionTrace.push({
        step: iteration,
        toolCall,
        result,
        timestamp: new Date().toISOString()
      });
    }
    
    // Step 4: Re-generate with tool results
    const toolResults = toolExecutionTrace.filter(t => t.step === iteration);
    const followUpPrompt = `${prompt}\n\nTool results: ${JSON.stringify(toolResults)}\nPlease continue.`;
    
    candidate = await provider.generateWithTools({
      model: 'gemini-1.5-pro',
      prompt: followUpPrompt,
      maxTokens: req.maxTokens,
      temperature: req.temperature
    }, tools);
  }
  
  const responseText = candidate.text || '';
  
  const context: OrchestratorContext = {
    identity_anchor,
    queryEmbedding: embedding,
    relevantMemories: memories,
    systemPrompt,
    fullPrompt: prompt,
    selectedProvider: provider.name
  };
  
  // Step 5: Reflective evaluation
  const reflectiveResult = this.reflectiveService.evaluate({ text: responseText }, memories);
  
  // Return response with toolExecutionTrace embedded in context
  return {
    id: candidate.id || `res_${Date.now()}`,
    text: responseText,
    context: {
      ...context,
      toolExecutionTrace
    } as any,
    reflectiveResult
  };
}
```
