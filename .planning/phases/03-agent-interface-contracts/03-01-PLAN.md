export interface ReasonRequest {
  identity_anchor: string;
  messages: Array<{ role: string; content: string }>;
  options?: {
    maxTokens?: number;
    temperature?: number;

    memoryAlpha?: number;
    memoryK?: number;
    skipReflection?: boolean;
    enableTools?: boolean;
    mode?: "tool" | "task";
    maxToolIterations?: number;
    maxTaskIterations?: number;
  };
}

export interface ReasonResponse {
  id: string;
  output: string;
  status: "approved" | "refine" | "reject";

      selectedProvider: string;

    reflective: ReflectiveCheckResult;
    trace: ReasoningTrace[];
  };
  toolCalls?: ToolCall[];
  toolResults?: ToolExecutionResult[];
  toolExecutionTrace?: ToolExecutionTrace[];
  plan?: TaskPlan;
  actions?: Action[];

}

<task type="auto">
  <name>task 1: Create OpenAPI 3.0 specification</name>
  <files>api/openapi.yml</files>
  <action>
Create `api/openapi.yml` with OpenAPI 3.0.3 specification covering all agent endpoints:

1. **Info block**: title "Aetherium Agent API", version "1.0.0", description linking to project docs
2. **Servers**: `http://localhost:8080` with description "Local development"
3. **Paths**:
   - `POST /v1/reason` — Natural language reasoning endpoint
     - Request body: ReasonRequest schema (fully specified with all option fields)
     - Responses: 200 (ReasonResponse), 400 (ValidationError), 500 (InternalError)
   - `GET /health` — Health check endpoint
     - Response: 200 with status, timestamp, version
   - `GET /v1/info` — API info endpoint (to be implemented in Plan 03-03)
     - Response: 200 with apiVersion, supportedVersions, features

4. **Components/Schemas** (fully specify every field from existing types):
   - ReasonRequest: identity_anchor (string, required), messages (array of Message, required), options (ReasonOptions, optional)
   - Message: role (enum: user, system, assistant), content (string)
   - ReasonOptions: maxTokens, temperature, policy, memoryAlpha, memoryK, skipReflection, enableTools, mode (enum: tool, task), maxToolIterations, maxTaskIterations — all optional with defaults documented
   - ReasonResponse: id, output, status (enum), identity_anchor, reasoning (Reasoning object), toolCalls, toolResults, toolExecutionTrace, plan, actions, planStatus, metadata — all as per existing type except metadata.apiVersion is required string "1.0.0"
   - Reasoning: orchestrator (OrchestratorInfo), reflective (ReflectiveResult), trace (array of ReasoningTrace)
   - OrchestratorInfo: selectedProvider, relevantMemoriesCount, topMemories
   - ReflectiveResult: status (enum: approved, refine, reject), violations (array), suggestedConstraints (array), confidenceScore (number 0-1)
   - ReasoningTrace: stage, timestamp, details (any object)
   - TaskPlan: steps (array of PlanStep), description, estimatedSteps
   - PlanStep: stepNumber, tool, args (object), expectedResult, status (enum), retryCount
   - Action: type (enum: report, update, trigger, notify), title, data (any), format (string)
   - ProblemDetails (RFC 7807): type (URI string), title, status (HTTP code), detail, instance (request ID), errors (array of {field, message})

5. **Security**: None for v1 (documented as "No authentication required for v1")

Use YAML format. Every schema property must have `type`, `description`, and `example` where applicable.
